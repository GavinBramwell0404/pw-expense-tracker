  <script>

    const LS_KEY = "pw_expenses_v1";

    const dateEl = document.getElementById("date");

    const vendorEl = document.getElementById("vendor");

    const categoryEl = document.getElementById("category");

    const amountEl = document.getElementById("amount");

    const notesEl = document.getElementById("notes");

    const formEl = document.getElementById("expenseForm");

    const monthFilterEl = document.getElementById("monthFilter");

    const categoryFilterEl = document.getElementById("categoryFilter");

    const searchEl = document.getElementById("search");

    const totalBoxEl = document.getElementById("totalBox");

    const summaryBoxesEl = document.getElementById("summaryBoxes");

    const expenseListEl = document.getElementById("expenseList");

    const countBoxEl = document.getElementById("countBox");

    const exportBtn = document.getElementById("exportBtn");

    const clearBtn = document.getElementById("clearBtn");


    let items = JSON.parse(localStorage.getItem(LS_KEY) || "[]");


    function todayISO() {

      return new Date().toISOString().slice(0, 10);

    }


    function fmtUSD(n) {

      return Number(n || 0).toLocaleString(undefined, { style: "currency", currency: "USD" });

    }


    function monthKey(date) {

      return (date || "").slice(0, 7);

    }


    function save() {

      localStorage.setItem(LS_KEY, JSON.stringify(items));

    }


    function getFiltered() {

      const month = monthFilterEl.value;

      const cat = categoryFilterEl.value;

      const q = searchEl.value.trim().toLowerCase();


      return items

        .filter(i => month === "all" || monthKey(i.date) === month)

        .filter(i => cat === "all" || i.category === cat)

        .filter(i => {

          if (!q) return true;

          return `${i.vendor} ${i.category} ${i.notes} ${i.date}`.toLowerCase().includes(q);

        })

        .sort((a, b) => b.date.localeCompare(a.date));

    }


    function refreshMonths() {

      const current = monthFilterEl.value;

      const months = [...new Set(items.map(i => monthKey(i.date)).filter(Boolean))].sort().reverse();

      monthFilterEl.innerHTML = '<option value="all">All</option>';

      months.forEach(m => {

        const opt = document.createElement("option");

        opt.value = m;

        opt.textContent = m;

        monthFilterEl.appendChild(opt);

      });

      monthFilterEl.value = months.includes(current) ? current : "all";

    }


    function render() {

      refreshMonths();

      const filtered = getFiltered();

      const total = filtered.reduce((sum, i) => sum + Number(i.amount || 0), 0);

      totalBoxEl.textContent = fmtUSD(total);

      countBoxEl.textContent = filtered.length;


      const byCat = {};

      filtered.forEach(i => {

        byCat[i.category] = (byCat[i.category] || 0) + Number(i.amount || 0);

      });


      summaryBoxesEl.innerHTML = "";

      const topCats = Object.keys(byCat).sort((a, b) => byCat[b] - byCat[a]).slice(0, 6);

      if (topCats.length === 0) {

        summaryBoxesEl.innerHTML = '<div class="mini muted">No expenses yet.</div>';

      } else {

        topCats.forEach(cat => {

          const div = document.createElement("div");

          div.className = "mini";

          div.innerHTML = `<strong>${cat}</strong><div class="muted">${fmtUSD(byCat[cat])}</div>`;

          summaryBoxesEl.appendChild(div);

        });

      }


      expenseListEl.innerHTML = "";

      if (filtered.length === 0) {

        expenseListEl.innerHTML = '<div class="empty">No matching expenses.</div>';

        return;

      }


      filtered.forEach(item => {

        const row = document.createElement("div");

        row.className = "expense-item";

        row.innerHTML = `

          <div class="expense-main">

            <div class="expense-top">

              <div class="expense-title">${escapeHtml(item.vendor)}</div>

              <div><strong>${fmtUSD(item.amount)}</strong></div>

            </div>

            <div class="expense-meta">

              <span>${escapeHtml(item.date)}</span>

              <span>•</span>

              <span>${escapeHtml(item.category)}</span>

              ${item.notes ? `<span>•</span><span>${escapeHtml(item.notes)}</span>` : ""}

            </div>

          </div>

          <button class="delete-btn" data-id="${item.id}">Delete</button>

        `;

        expenseListEl.appendChild(row);

      });

    }


    function escapeHtml(str) {

      return String(str || "")

        .replaceAll("&", "&amp;")

        .replaceAll("<", "&lt;")

        .replaceAll(">", "&gt;")

        .replaceAll('"', "&quot;")

        .replaceAll("'", "&#039;");

    }


    function exportCSV() {

      const filtered = getFiltered();

      const rows = [

        ["Date", "Vendor", "Category", "Amount", "Notes"],

        ...filtered.map(i => [i.date, i.vendor, i.category, i.amount, i.notes || ""])

      ];

      const csv = rows.map(r => r.map(v => {

        const s = String(v ?? "");

        return /[\n\r,"]/.test(s) ? `"${s.replace(/"/g, '""')}"` : s;

      }).join(",")).join("\n");


      const blob = new Blob([csv], { type: "text/csv;charset=utf-8" });

      const url = URL.createObjectURL(blob);

      const a = document.createElement("a");

      a.href = url;

      a.download = "expenses.csv";

      a.click();

      URL.revokeObjectURL(url);

    }


    formEl.addEventListener("submit", function (e) {

      e.preventDefault();

      const date = dateEl.value;

      const vendor = vendorEl.value.trim();

      const category = categoryEl.value;

      const amount = Number(amountEl.value);

      const notes = notesEl.value.trim();


      if (!date || !vendor || !Number.isFinite(amount) || amount <= 0) return;


      items.unshift({

        id: String(Date.now() + Math.random()),

        date,

        vendor,

        category,

        amount: Math.round(amount * 100) / 100,

        notes

      });


      save();

      vendorEl.value = "";

      amountEl.value = "";

      notesEl.value = "";

      render();

    });


    expenseListEl.addEventListener("click", function (e) {

      if (!e.target.matches(".delete-btn")) return;

      const id = e.target.getAttribute("data-id");

      items = items.filter(i => i.id !== id);

      save();

      render();

    });


    [monthFilterEl, categoryFilterEl, searchEl].forEach(el => {

      el.addEventListener("input", render);

      el.addEventListener("change", render);

    });


    exportBtn.addEventListener("click", exportCSV);

    clearBtn.addEventListener("click", function () {

      if (!confirm("Clear all saved expenses?")) return;

      items = [];

      save();

      render();

    });


    dateEl.value = todayISO();

    render();

  </script>
